function doPost(e) {

  var headers = {
    "Access-Control-Allow-Origin": "*",
    "Access-Control-Allow-Methods": "POST, GET, OPTIONS",
    "Access-Control-Allow-Headers": "Content-Type"
  };

  try {
    var data = JSON.parse(e.postData.contents);

    // ── 1. Get or Create Google Drive Folder for Photos ──────────────────────
    var folderName = "StickerStars_Photos";
    var folders = DriveApp.getFoldersByName(folderName);
    var folder = folders.hasNext() ? folders.next() : DriveApp.createFolder(folderName);

    // ── 2. Decode and Save Photo to Google Drive ──────────────────────────────
    var photoUrl = "No Photo Uploaded";
    if (data.photoData) {
      var contentType = data.photoType || "image/jpeg";
      var decodedPhoto = Utilities.base64Decode(data.photoData);
      var filename = (data.name || "unknown").replace(/[^a-z0-9]/gi, '_').toLowerCase() + "_photo";
      var blob = Utilities.newBlob(decodedPhoto, contentType, filename);
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      photoUrl = file.getUrl();
    }

    // ── 3. Append Details to Google Sheet ─────────────────────────────────────
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var timestamp = new Date();

    if (sheet.getLastRow() === 0) {
      sheet.appendRow([
        "Timestamp",
        "Child Name",
        "Class/Grade",
        "Division/Section",
        "Roll No",
        "Subject",
        "School Name",
        "WhatsApp Number",
        "Photo Google Drive Link",
        "Theme & Special Requests"
      ]);
    }

    sheet.appendRow([
      timestamp,
      data.name      || "",
      data.className || "",
      data.division  || "",
      data.roll      || "N/A",
      data.subject   || "N/A",
      data.school    || "",
      data.whatsapp  || "",
      photoUrl,
      data.requests  || ""
    ]);

    // ── 4. Build HTML Email ───────────────────────────────────────────────────
    var photoSection = (photoUrl !== "No Photo Uploaded")
      ? '<a href="' + photoUrl + '" style="display:inline-block;margin-top:8px;padding:10px 24px;background:#ff3d8a;color:#ffffff;text-decoration:none;border-radius:25px;font-size:14px;font-weight:600;">📷 View Child\'s Photo</a>'
      : '<span style="color:#999;font-size:13px;">No photo uploaded</span>';

    var htmlBody =
      '<!DOCTYPE html>' +
      '<html><head><meta charset="UTF-8"></head>' +
      '<body style="margin:0;padding:0;background-color:#f9f0f5;font-family:Arial,sans-serif;">' +

        // ── Wrapper
        '<table width="100%" cellpadding="0" cellspacing="0" style="background:#f9f0f5;padding:30px 0;">' +
          '<tr><td align="center">' +
            '<table width="600" cellpadding="0" cellspacing="0" style="background:#ffffff;border-radius:16px;overflow:hidden;box-shadow:0 4px 20px rgba(255,61,138,0.12);">' +

              // ── Header
              '<tr>' +
                '<td style="background:linear-gradient(135deg,#ff3d8a,#ff85b3);padding:32px 40px;text-align:center;">' +
                  '<div style="display:inline-flex;align-items:center;gap:10px;">' +
                    '<span style="background:#fff;border-radius:50%;width:42px;height:42px;display:inline-block;line-height:42px;font-size:22px;">⭐</span>' +
                    '<span style="color:#fff;font-size:26px;font-weight:800;letter-spacing:-0.5px;">MagicLabel</span>' +
                  '</div>' +
                  '<p style="color:#ffe0ee;margin:10px 0 0;font-size:14px;">Personalized Labels for Every Little Star</p>' +
                '</td>' +
              '</tr>' +

              // ── Alert Banner
              '<tr>' +
                '<td style="background:#fff8fb;padding:20px 40px;border-bottom:2px dashed #ffcce0;text-align:center;">' +
                  '<p style="margin:0;font-size:22px;font-weight:700;color:#1a1a2e;">🎉 New Order Received!</p>' +
                  '<p style="margin:6px 0 0;font-size:14px;color:#888;">Someone just placed a label order for <strong style="color:#ff3d8a;">' + (data.name || "Unknown") + '</strong></p>' +
                '</td>' +
              '</tr>' +

              // ── Order Details
              '<tr>' +
                '<td style="padding:30px 40px;">' +
                  '<p style="margin:0 0 16px;font-size:13px;font-weight:700;color:#ff3d8a;text-transform:uppercase;letter-spacing:1px;">Order Details</p>' +

                  '<table width="100%" cellpadding="0" cellspacing="0">' +

                    // Row helper — alternating background
                    '<tr style="background:#fff8fb;">' +
                      '<td style="padding:12px 16px;font-size:13px;color:#888;width:40%;border-radius:8px 0 0 8px;">👦 Child Name</td>' +
                      '<td style="padding:12px 16px;font-size:14px;font-weight:700;color:#1a1a2e;border-radius:0 8px 8px 0;">' + (data.name || "—") + '</td>' +
                    '</tr>' +
                    '<tr><td colspan="2" style="height:6px;"></td></tr>' +

                    '<tr style="background:#fff8fb;">' +
                      '<td style="padding:12px 16px;font-size:13px;color:#888;border-radius:8px 0 0 8px;">🏫 School</td>' +
                      '<td style="padding:12px 16px;font-size:14px;font-weight:700;color:#1a1a2e;border-radius:0 8px 8px 0;">' + (data.school || "—") + '</td>' +
                    '</tr>' +
                    '<tr><td colspan="2" style="height:6px;"></td></tr>' +

                    '<tr style="background:#fff8fb;">' +
                      '<td style="padding:12px 16px;font-size:13px;color:#888;border-radius:8px 0 0 8px;">📚 Class & Division</td>' +
                      '<td style="padding:12px 16px;font-size:14px;font-weight:700;color:#1a1a2e;border-radius:0 8px 8px 0;">' + (data.className || "—") + ' &nbsp;–&nbsp; ' + (data.division || "—") + '</td>' +
                    '</tr>' +
                    '<tr><td colspan="2" style="height:6px;"></td></tr>' +

                    '<tr style="background:#fff8fb;">' +
                      '<td style="padding:12px 16px;font-size:13px;color:#888;border-radius:8px 0 0 8px;">🔢 Roll No</td>' +
                      '<td style="padding:12px 16px;font-size:14px;font-weight:700;color:#1a1a2e;border-radius:0 8px 8px 0;">' + (data.roll || "N/A") + '</td>' +
                    '</tr>' +
                    '<tr><td colspan="2" style="height:6px;"></td></tr>' +

                    '<tr style="background:#fff8fb;">' +
                      '<td style="padding:12px 16px;font-size:13px;color:#888;border-radius:8px 0 0 8px;">📖 Subject</td>' +
                      '<td style="padding:12px 16px;font-size:14px;font-weight:700;color:#1a1a2e;border-radius:0 8px 8px 0;">' + (data.subject || "N/A") + '</td>' +
                    '</tr>' +
                    '<tr><td colspan="2" style="height:6px;"></td></tr>' +

                    '<tr style="background:#fff8fb;">' +
                      '<td style="padding:12px 16px;font-size:13px;color:#888;border-radius:8px 0 0 8px;">📱 WhatsApp</td>' +
                      '<td style="padding:12px 16px;font-size:14px;font-weight:700;color:#1a1a2e;border-radius:0 8px 8px 0;">' + (data.whatsapp || "—") + '</td>' +
                    '</tr>' +
                    '<tr><td colspan="2" style="height:6px;"></td></tr>' +

                    '<tr style="background:#fff8fb;">' +
                      '<td style="padding:12px 16px;font-size:13px;color:#888;border-radius:8px 0 0 8px;">💬 Requests</td>' +
                      '<td style="padding:12px 16px;font-size:14px;font-weight:700;color:#1a1a2e;border-radius:0 8px 8px 0;">' + (data.requests || "None") + '</td>' +
                    '</tr>' +

                  '</table>' +

                  // ── Photo Button
                  '<div style="margin-top:24px;text-align:center;">' +
                    photoSection +
                  '</div>' +

                '</td>' +
              '</tr>' +

              // ── CTA Button
              '<tr>' +
                '<td style="padding:0 40px 30px;text-align:center;">' +
                  '<a href="https://docs.google.com/spreadsheets" style="display:inline-block;padding:14px 36px;background:linear-gradient(135deg,#ff3d8a,#ff85b3);color:#ffffff;text-decoration:none;border-radius:30px;font-size:15px;font-weight:700;box-shadow:0 4px 15px rgba(255,61,138,0.35);">📊 Open Google Sheet</a>' +
                '</td>' +
              '</tr>' +

              // ── Footer
              '<tr>' +
                '<td style="background:#f9f0f5;padding:20px 40px;text-align:center;border-top:1px solid #ffe0ee;">' +
                  '<p style="margin:0;font-size:12px;color:#bbb;">This is an automated notification from <strong style="color:#ff3d8a;">MagicLabel</strong></p>' +
                  '<p style="margin:6px 0 0;font-size:12px;color:#ccc;">© 2025 MagicLabel · Personalized Labels for Every Little Star ⭐</p>' +
                '</td>' +
              '</tr>' +

            '</table>' +
          '</td></tr>' +
        '</table>' +

      '</body></html>';

    // ── 5. Send HTML Email ────────────────────────────────────────────────────
    var emailRecipient = Session.getEffectiveUser().getEmail();

    MailApp.sendEmail({
      to       : emailRecipient,
      subject  : "⭐ New MagicLabel Order — " + (data.name || "Unknown"),
      body     : "New order for " + (data.name || "Unknown") + " from " + (data.school || "Unknown School"),  // plain text fallback
      htmlBody : htmlBody
    });

    // ── 6. Return Success Response ────────────────────────────────────────────
    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", photoUrl: photoUrl }))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (error) {

    // ── Return Error Response ─────────────────────────────────────────────────
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
